# MississippiFields

State-specific fields for Mississippi TAP portal registration import.

## Example Usage

```typescript
import { MississippiFields } from "@kintsugi-tax/tax-platform-sdk/models";

let value: MississippiFields = {
  registrationType: "SALES_AND_USE_TAX",
  businessName: "<value>",
  msStateTaxId: "<id>",
  msAccountType: "SALES_TAX_LICENSE",
  letterId: "<id>",
};
```
