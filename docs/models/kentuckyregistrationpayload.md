# KentuckyRegistrationPayload

Full payload for Kentucky registration import.

## Example Usage

```typescript
import { KentuckyRegistrationPayload } from "@kintsugi-tax/tax-platform-sdk/models";

let value: KentuckyRegistrationPayload = {
  countryCode: "QA",
  stateCode: "<value>",
  stateName: "<value>",
  jurisdictionSpecificFields: {
    registrationType: "SALES_AND_USE_TAX",
    businessName: "<value>",
    kyStateTaxId: "<id>",
  },
};
```
